from typing import List

def below_zero(operations: List[int]) -> bool:
    """
    Detect if the balance of the bank account falls below zero at any point during the operations.

    Args:
        operations (List[int]): A list of deposit and withdrawal operations on the bank account.

    Returns:
        bool: True if the balance of the account falls below zero at any point, False otherwise.
    """
    balance = 0
    for op in operations:
        balance += op
        if balance < 0:
            return True
    return False
